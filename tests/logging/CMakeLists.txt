cmake_minimum_required(VERSION 3.21)
project(ConnectionLoggingTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Test Gui Network Core5Compat)
set(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../..")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version.h" "#define APPLICATION_NAME \"LoggingTest\"\n#define SERVICE_NAME \"LoggingTest-service\"\n")
add_executable(connection_logging main.cpp "${REPO}/common/logger/logger.cpp" "${REPO}/common/logger/logger.h")
target_include_directories(connection_logging PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${REPO}/common/logger" "${REPO}/client")
target_compile_definitions(connection_logging PRIVATE DEV_AGW_ENDPOINT="http://127.0.0.1/")
target_link_libraries(connection_logging PRIVATE Qt6::Test Qt6::Gui Qt6::Network Qt6::Core5Compat)
enable_testing()
add_test(NAME connection_logging COMMAND connection_logging)
