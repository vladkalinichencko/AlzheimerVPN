cmake_minimum_required(VERSION 3.21)
project(RequestDeadlineTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Test Network)
set(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(request_deadline main.cpp "${REPO}/client/core/utils/api/requestDeadline.h")
target_include_directories(request_deadline PRIVATE "${REPO}/client")
target_link_libraries(request_deadline PRIVATE Qt6::Test Qt6::Network)
enable_testing()
add_test(NAME request_deadline COMMAND request_deadline)
